## 3. Next.js Route Handler (`app/api/send/route.ts`)

```typescript
import { NextResponse } from 'next/server';
import { Resend } from 'resend';
import { WelcomeEmail } from '@/emails/WelcomeEmail';

// Initialized with process.env.RESEND_BASE_URL pointing to EmitMail
const resend = new Resend(process.env.RESEND_API_KEY);

export async function POST(req: Request) {
  try {
    const { email, name } = await req.json();

    const { data, error } = await resend.emails.send({
      from: 'EmitMail Team <hello@yourdomain.com>',
      to: [email],
      subject: 'Your sovereign email workspace is active',
      react: WelcomeEmail({ name, loginUrl: 'https://yourdomain.com/login' }),
    });

    if (error) {
      return NextResponse.json({ error }, { status: 400 });
    }

    return NextResponse.json({ success: true, id: data?.id });
  } catch (err: any) {
    return NextResponse.json({ error: err.message }, { status: 500 });
  }
}
```
